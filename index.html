create table entries (id text primary key, type text, amount numeric, category text, date date, note text, created_at timestamp default now()); alter table entries enable row level security; create policy "public access" on entries for all using (true); create table budgets (id text primary key, category text, monthly_limit numeric, created_at timestamp default now()); alter table budgets enable row level security; create policy "public access" on budgets for all using (true);
